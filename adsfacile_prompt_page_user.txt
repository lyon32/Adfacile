PARTIE 1 - QUELLE API IA UTILISER ?
🎯 Contexte d'utilisation
AdsFacile utilise l'IA pour 3 choses précises :
  1. Génération de textes publicitaires (3 variations)
  2. Propositions de ciblage (audience, âge, intérêts) selon la niche
  3. Analyse de texte / contenu pour recommandations (plan PRO/Agency)

Ces appels IA sont courts (300-800 tokens input, 400-600 tokens output).
Volume estimé : 50-500 appels/jour en phase bêta, 5000+/jour à l'échelle.

1.2 Configuration API Gemini (Gratuit / Plan Free)
// .env
GEMINI_API_KEY=votre_cle_google_ai_studio
GEMINI_MODEL=gemini-1.5-flash

// services/aiService.js
const { GoogleGenerativeAI } = require('@google/generative-ai');
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);

async function generateAdText({ product, niche, audience, tone }) {
  const model = genAI.getGenerativeModel({ model: 'gemini-1.5-flash' });
  const prompt = `Tu es expert en publicités Facebook pour PME africaines.
  Génère 3 textes publicitaires pour :
  - Produit/Service : ${product}
  - Secteur : ${niche}
  - Audience cible : ${audience}
  - Ton souhaité : ${tone}
  Format JSON : [{text, headline, cta}]`;
  const result = await model.generateContent(prompt);
  return JSON.parse(result.response.text());
}

// Coût estimé : $0 jusqu'à 1500 req/jour (quota gratuit Google)
1.3 Configuration API Claude Haiku (Plan PRO/Agency)
// .env
ANTHROPIC_API_KEY=votre_cle_anthropic
CLAUDE_MODEL=claude-haiku-4-5-20251001

// services/aiServicePro.js
const Anthropic = require('@anthropic-ai/sdk');
const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

async function generateAdTextPro({ product, niche, audience, budget, city }) {
  const msg = await client.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 1024,
    messages: [{
      role: 'user',
      content: `Expert Facebook Ads Afrique. Génère 3 variations pub pour:
      Business: ${product} | Niche: ${niche}
      Ville: ${city} | Budget: ${budget} FCFA/jour
      Audience: ${audience}
      Inclure: texte, titre, CTA, ciblage suggéré (âge, intérêts)
      Format: JSON valide uniquement.`
    }]
  });
  return JSON.parse(msg.content[0].text);
}

// Coût estimé : ~292 FCFA / 1000 appels
